#include <iostream>
#include <algorithm>

using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    for (int test = 1; test <= T; test++) {
        long long N, A, B;
        cin >> N >> A >> B;

        long long ans = 1e18;

        for (long long r = 1; r * r <= N; r++) {

            long long c;

            // 비용 = (A - B*r)c + BN - Ar
            if (A - B * r >= 0) c = r;
            else c = N / r;
            
            long long cost = A * (c - r) + B * (N - r * c);
            ans = min(ans, cost);
        }

        cout << "#" << test << " " << ans << '\n';
    }

    return 0;
}
